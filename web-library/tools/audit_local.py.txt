import hashlib
import importlib.util
import json
import os
import shutil
import subprocess
import sys
from collections import Counter
from datetime import datetime, timezone
from html.parser import HTMLParser
from pathlib import Path


ROOT = Path(__file__).parent.parent.parent
INPUTS = ["Index.html.txt", "build.py.txt", "src/page.html.txt", "src/styles.css.txt",
          "src/motion.css.txt", "src/app.js.txt", "src/effects.js.txt"]


class DocumentAudit(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.identifiers = []
        self.anchors = []
        self.aria_targets = []
        self.resource_references = []
        self.tags = Counter()
        self.language = None

    def handle_starttag(self, tag, attributes):
        values = dict(attributes)
        self.tags[tag] += 1
        if "id" in values:
            self.identifiers.append(values["id"])
        if tag == "html":
            self.language = values.get("lang")
        if tag == "a" and values.get("href", "").startswith("#"):
            self.anchors.append(values["href"][1:])
        for attribute in ("aria-labelledby", "aria-describedby", "aria-controls"):
            self.aria_targets.extend((values.get(attribute) or "").split())
        for attribute in ("src", "poster", "srcset"):
            if values.get(attribute):
                self.resource_references.append(values[attribute])
        if tag == "link" and values.get("href"):
            self.resource_references.append(values["href"])
        if tag in {"use", "image"}:
            for attribute in ("href", "xlink:href"):
                target = values.get(attribute, "")
                if target and not target.startswith("#"):
                    self.resource_references.append(target)

    def handle_startendtag(self, tag, attributes):
        self.handle_starttag(tag, attributes)


def main():
    sources = {name: (ROOT / name).read_text(encoding="utf-8") for name in INPUTS}
    assembled = sources["src/page.html.txt"]
    markers = {"{{STYLES}}": "\n".join(sources[name] for name in ("src/styles.css.txt", "src/motion.css.txt")),
               "{{SCRIPT}}": "\n".join(sources[name] for name in ("src/app.js.txt", "src/effects.js.txt"))}
    marker_counts = {marker: assembled.count(marker) for marker in markers}
    for marker, replacement in markers.items():
        assembled = assembled.replace(marker, replacement)
    document = DocumentAudit()
    document.feed(sources["Index.html.txt"])
    identifiers = set(document.identifiers)
    duplicates = sorted(identifier for identifier, count in Counter(document.identifiers).items() if count > 1)
    checks = []

    def record(identifier, passed, evidence):
        checks.append({"id": identifier, "status": "pass" if passed else "fail", "evidence": evidence})

    record("assembly_equal", assembled.encode("utf-8") == (ROOT / "Index.html.txt").read_bytes()
           and all(count == 1 for count in marker_counts.values()),
           {"method": "Reproduce reviewed substitutions in memory; do not execute or overwrite build.py.txt",
            "markers": marker_counts, "equal_after_newline_normalization": assembled == sources["Index.html.txt"],
            "equal_nonblank_lines": [line for line in assembled.splitlines() if line.strip()]
            == [line for line in sources["Index.html.txt"].splitlines() if line.strip()],
            "reconstructed_utf8_bytes": len(assembled.encode("utf-8"))})
    record("unique_ids", not duplicates, {"count": len(document.identifiers), "duplicates": duplicates})
    missing_anchors = sorted(set(document.anchors) - identifiers)
    record("anchor_targets", not missing_anchors, {"count": len(document.anchors), "missing": missing_anchors})
    missing_aria = sorted(set(document.aria_targets) - identifiers)
    record("aria_idrefs", not missing_aria, {"missing": missing_aria})
    record("document_basics", document.tags["h1"] == 1 and document.tags["main"] == 1 and bool(document.language),
           {"h1": document.tags["h1"], "main": document.tags["main"], "sections": document.tags["section"], "lang": document.language})
    css_text = sources["src/styles.css.txt"] + "\n" + sources["src/motion.css.txt"]
    if importlib.util.find_spec("tinycss2"):
        import tinycss2

        def collect_urls(nodes):
            references = []
            for node in nodes:
                if node.type == "url":
                    references.append(node.value)
                elif node.type == "function" and node.lower_name == "url":
                    arguments = [argument for argument in node.arguments if argument.type != "whitespace"]
                    if len(arguments) == 1 and arguments[0].type == "string":
                        references.append(arguments[0].value)
                    else:
                        references.append(tinycss2.serialize(node.arguments).strip())
                else:
                    references.extend(collect_urls(getattr(node, "content", [])))
                    references.extend(collect_urls(getattr(node, "arguments", [])))
            return references

        css_references = collect_urls(tinycss2.parse_component_value_list(css_text))
        references = document.resource_references + [target for target in css_references if not target.startswith(("#", "data:"))]
        record("static_resource_references", not references,
               {"references": references, "scope": "HTML resource attributes and CSS url() tokens; not JS or all possible CSS resource syntax"})
        errors = []
        for name in ("src/styles.css.txt", "src/motion.css.txt"):
            nodes = tinycss2.parse_stylesheet(sources[name], skip_comments=True, skip_whitespace=True)
            errors.extend({"file": name, "message": node.message} for node in nodes if node.type == "error")
        record("css_basic_parse", not errors, {"errors": errors, "scope": "Top-level parser only; not semantic CSS validation or rendering"})
    else:
        checks.append({"id": "css_basic_parse", "status": "not_run", "evidence": "tinycss2 is not installed"})
        record("html_resource_references", not document.resource_references,
               {"references": document.resource_references, "scope": "HTML only; CSS and JavaScript not checked"})
    if shutil.which("deno"):
        for name in ("src/app.js.txt", "src/effects.js.txt"):
            command = ["deno", "eval", "--no-config", "--no-lock",
                       'new Function(await new Response(Deno.stdin.readable).text()); console.log("Parsed, not executed");']
            try:
                result = subprocess.run(command, input=sources[name], text=True, capture_output=True,
                                        timeout=20, cwd=ROOT, env={**os.environ, "DENO_NO_UPDATE_CHECK": "1"})
                record("js_parse:" + name, result.returncode == 0,
                       {"stdout": result.stdout.strip(), "stderr": result.stderr.strip(),
                        "scope": "Existing IIFE parsed as Function body, not executed; no DOM/browser check"})
            except subprocess.TimeoutExpired:
                checks.append({"id": "js_parse:" + name, "status": "not_run", "evidence": "Deno timed out"})
    else:
        checks.append({"id": "js_parse", "status": "not_run", "evidence": "Deno unavailable"})
    for identifier in ("browser_render", "keyboard_behavior", "responsive_visual", "reduced_motion_behavior",
                       "offline_browser_launch", "performance_measurement", "human_design_review"):
        checks.append({"id": identifier, "status": "not_run", "evidence": "Not performed by this static audit"})
    report = {
        "scope": "Existing local site; not a baseline generation and not a quality benchmark",
        "checked_at": datetime.now(timezone.utc).isoformat(),
        "input_sha256": {name: hashlib.sha256((ROOT / name).read_bytes()).hexdigest() for name in INPUTS},
        "html_bytes": (ROOT / "Index.html.txt").stat().st_size,
        "checks": checks,
        "summary": dict(Counter(check["status"] for check in checks)),
        "files_changed_by_audit": ["web-library/reports/local-audit.json"]
    }
    target = ROOT / "web-library/reports/local-audit.json"
    if not target.resolve().is_relative_to(ROOT.resolve()):
        raise ValueError("Audit report path escapes workspace")
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps({"report": "web-library/reports/local-audit.json", "summary": report["summary"]}, ensure_ascii=False))
    return 1 if any(check["status"] == "fail" for check in checks) else 0


if __name__ == "__main__":
    sys.exit(main())
